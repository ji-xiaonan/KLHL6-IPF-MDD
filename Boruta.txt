install.packages("Boruta")
install.packages("mlbench")
rm(list=ls())
library(Boruta)
library(mlbench)
library(tidyverse)
Boruta_input = read.table("train.txt",sep = "\t",check.names = F,stringsAsFactors = F,header = T,row.names = 1)

Boruta_input$status <- as.factor(Boruta_input$status)
set.seed(1)
boruta <- Boruta(status ~ ., data = Boruta_input, doTrace = 2, maxRuns = 500)

print(boruta)
# 重要变量对应下图绿色，不重要的变量对应下图红色，不明确重要变量对应下图黄色

#可视化选择的结果
pdf("A_Boruta.pdf",width = 5, height = 5)
plot(boruta, las = 2, cex.axis = 0.7)
dev.off()
#可视化具体的选择过程
pdf("B_Boruta.pdf", width = 5, height = 5)
plotImpHistory(boruta)
dev.off()

#采用TentativeRoughFix的方法来解析不确定的变量
bor <- TentativeRoughFix(boruta)
print(bor)
plot(bor, las = 2, cex.axis = 0.7)

#显示最终变量选择的结果
bbb <- attStats(boruta)
write.table(attStats(boruta), "output_importance_Boruta.txt", col.names = T, row.names = T, sep = "\t", quote = F)

gene4 <- rownames(bbb)[which(bbb$decision == "Confirmed")]
